# Fitzhugh protectorate flag picker.
# Caller: COR_establish_protectorate_<TAG> complete_effect on COR (player only).
# ROOT = COR. Options apply set_cosmetic_tag on the released protectorate.
# One event per tag that has extra COR_<TAG>_OCC_<N> flags.

add_namespace = KXWD_COR_fitzhugh_flag

country_event = {
	id = KXWD_COR_fitzhugh_flag.1
	title = KXWD_COR_fitzhugh_flag.1.t
	desc = KXWD_COR_fitzhugh_flag.1.desc
	picture = GFX_evt_COR_BHC_OCC_flags
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event KXWD_COR_fitzhugh_flag.1 (BHC flag picker)"
	}

	option = {
		name = KXWD_COR_fitzhugh_flag.1.a
		custom_effect_tooltip = KXWD_COR_fitzhugh_flag.1.a.tt
		if = {
			limit = { BHC = { exists = yes } }
			BHC = { set_cosmetic_tag = COR_BHC_OCC }
		}
	}
	option = {
		name = KXWD_COR_fitzhugh_flag.1.b
		custom_effect_tooltip = KXWD_COR_fitzhugh_flag.1.b.tt
		if = {
			limit = { BHC = { exists = yes } }
			BHC = { set_cosmetic_tag = COR_BHC_OCC_2 }
		}
	}
	option = {
		name = KXWD_COR_fitzhugh_flag.1.c
		custom_effect_tooltip = KXWD_COR_fitzhugh_flag.1.c.tt
		if = {
			limit = { BHC = { exists = yes } }
			BHC = { set_cosmetic_tag = COR_BHC_OCC_3 }
		}
	}
	option = {
		name = KXWD_COR_fitzhugh_flag.1.d
		custom_effect_tooltip = KXWD_COR_fitzhugh_flag.1.d.tt
		if = {
			limit = { BHC = { exists = yes } }
			BHC = { set_cosmetic_tag = COR_BHC_OCC_4 }
		}
	}
}
