			ideology = pataut_bonapartism_subtype
			traits = {
				RUS_tsar_of_all_russia_1
				ARM_armenian_guide
				ARM_national_savior
			}
		}
		set_party_name = {
    		ideology = paternal_autocrat

		set_cosmetic_tag = ARMWD_RUS_ARM_empire_OCC
	}
	option = {
		name = KXWD_ARM.2.b
		create_country_leader = {
			name = "Nzhdeh I"
			picture = "gfx/leaders/ARM/Portrait_Armenia_Garegin_Nzhdeh.png"
			expire = "1.1.1"
			ideology = socdem_bonapartism_subtype
			traits = {
				RUS_tsar_of_all_russia_1
				ARM_armenian_guide
				ARM_national_savior
			}
		}
		add_popularity = {
   			ideology = social_democrat
  		    popularity = 0.5
		}
		set_politics = {
    		ruling_party = social_democrat
    		elections_allowed = no
    		long_name = KXWD_ARM_RRF_long 
   			name = KXWD_ARM_RRF
		}
		swap_ideas = {
			remove_idea = KXWD_ARM_nzhdeh_bonaparte
			add_idea = KXWD_ARM_nzhdeh_bonaparte_socdem
		}
		set_cosmetic_tag = ARMWD_RUS_ARM_empire_OCC
	}
}
